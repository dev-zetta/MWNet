find_package(CURL 7.85 REQUIRED)
find_package(Boost 1.70 REQUIRED COMPONENTS system)
find_package(SQLite3 REQUIRED)
find_package(Threads REQUIRED)
add_library(mwnet-discovery STATIC Protocol.cpp Client.cpp)
target_include_directories(mwnet-discovery PUBLIC ${CMAKE_SOURCE_DIR})
target_link_libraries(mwnet-discovery PUBLIC mwnet-security Boost::system CURL::libcurl Threads::Threads)
add_library(mwnet-directory-core STATIC Directory.cpp Probe.cpp)
target_include_directories(mwnet-directory-core PUBLIC ${CMAKE_SOURCE_DIR})
target_link_libraries(mwnet-directory-core PUBLIC mwnet-discovery mwnet-game-endpoint SQLite::SQLite3)
if (MSVC)
    target_compile_options(mwnet-discovery PRIVATE /W4 /WX)
    target_compile_options(mwnet-directory-core PRIVATE /W4 /WX)
else()
    target_compile_options(mwnet-discovery PRIVATE -Wall -Wextra -Wpedantic -Werror)
    target_compile_options(mwnet-directory-core PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
